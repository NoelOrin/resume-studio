# Resume Studio 画布自由缩放 实现计划

> **面向 AI 代理的工作者：** 必需子技能：使用 superpowers:subagent-driven-development（推荐）或 superpowers:executing-plans 逐任务实现此计划。步骤使用复选框（`- [ ]`）语法来跟踪进度。

**目标：** 把预览画布从四个固定缩放档位升级为 25%–400% 连续缩放，并支持光标锚定滚轮缩放、拖拽平移、双击回适应与偏好持久化。

**架构：** 纯函数（缩放数学）→ `useZoom` hook（状态 + localStorage）→ `ZoomControl` 展示组件 → `EditorApp` 接线（原生非 passive wheel、pointer 平移、layout effect 锚点补偿）。CSS 同时修复居中容器溢出不可达问题。

**技术栈：** React 19 + TypeScript + vinext/Vite；测试用 `node:test` + `tsx`。

**规格：** `docs/superpowers/specs/2026-10-21-zoom-design.md`

```ts
// src/lib/zoom.ts
export const ZOOM_MIN = 0.25;
export const ZOOM_MAX = 4;
export const ZOOM_STEP = 0.1;

export type ZoomMode = "fit" | "manual";

export function clampZoom(value: number): number {
  if (!Number.isFinite(value)) return 1;
  return Math.min(ZOOM_MAX, Math.max(ZOOM_MIN, value));
}

export function snapZoom(value: number): number {
  return Math.round(clampZoom(value) * 100) / 100;
}

export function stepZoom(value: number, delta: number): number {
  return snapZoom(value + delta);
}

export function wheelZoom(value: number, deltaY: number): number {
  return snapZoom(value * Math.exp(-deltaY * 0.0015));
}

```

### 任务 2：useZoom hook（状态 + 持久化）

**文件：**
- 创建：`src/lib/useZoom.ts`

- [ ] **步骤 1：编写实现**

```ts
"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import { snapZoom, stepZoom as stepValue, type ZoomMode } from "./zoom";

const STORAGE_KEY = "resume-studio:zoom";

type Stored = { mode: ZoomMode; value: number };

function readStored(): Stored | null {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return null;
    const parsed = JSON.parse(raw) as Partial<Stored>;
    if (parsed.mode === "manual" && typeof parsed.value === "number") {
      return { mode: "manual", value: snapZoom(parsed.value) };
    }
    return { mode: "fit", value: 1 };
  } catch {
    return null;
  }
}

export function useZoom(fit: number) {
  const [mode, setMode] = useState<ZoomMode>("fit");
  const [value, setValue] = useState(1);
  const timer = useRef<number | undefined>(undefined);

  // restore after mount so SSR markup stays "fit"
  useEffect(() => {
    const stored = readStored();
    if (!stored) return;
    setMode(stored.mode);
    if (stored.mode === "manual") setValue(stored.value);
  }, []);

  const persist = useCallback((next: Stored) => {
    window.clearTimeout(timer.current);
    timer.current = window.setTimeout(() => {
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
      } catch {
        /* storage unavailable: keep in-memory */
      }
    }, 150);
  }, []);

  useEffect(() => () => window.clearTimeout(timer.current), []);

  const scale = mode === "fit" ? fit : value;

  const setManual = useCallback(
    (next: number) => {
      const snapped = snapZoom(next);
      setMode("manual");
      setValue(snapped);
      persist({ mode: "manual", value: snapped });
    },
    [persist],
  );

  const setFit = useCallback(() => {
    setMode("fit");
    persist({ mode: "fit", value: 1 });
  }, [persist]);

  const zoomBy = useCallback(
    (delta: number) => {
      setManual(stepValue(scale, delta));
    },
    [scale, setManual],
  );

  return { mode, scale, setManual, setFit, zoomBy };
}
```

```ts
const anchoredZoom = useCallback((next: number, clientX: number, clientY: number) => {
  const canvas = canvasRef.current;
  if (canvas) {
    const rect = canvas.getBoundingClientRect();
    anchorRef.current = {
      screenX: clientX,
      screenY: clientY,
      sheetX: (clientX - rect.left) / scale,
      sheetY: (clientY - rect.top) / scale,
    };
  }
  setManual(next);
}, [scale, setManual]);

useLayoutEffect(() => {
  scaleRef.current = scale;
  const anchor = anchorRef.current;
  anchorRef.current = null;
  if (!anchor) return;
  const stage = stageRef.current;
  const canvas = canvasRef.current;
  if (!stage || !canvas) return;
  const rect = canvas.getBoundingClientRect();
  stage.scrollLeft += rect.left + anchor.sheetX * scale - anchor.screenX;
  stage.scrollTop += rect.top + anchor.sheetY * scale - anchor.screenY;
}, [scale]);
```

- [ ] **步骤 3：原生非 passive 滚轮缩放**

```ts
useEffect(() => {
  const el = stageRef.current;
  if (!el) return;
  const onWheel = (event: WheelEvent) => {
    if (!(event.ctrlKey || event.metaKey)) return;
    event.preventDefault();
    wheelAccum.current += event.deltaY;
    wheelPoint.current = { x: event.clientX, y: event.clientY };
    if (wheelRaf.current !== undefined) return;
    wheelRaf.current = window.requestAnimationFrame(() => {
      wheelRaf.current = undefined;
      const deltaY = wheelAccum.current;
      wheelAccum.current = 0;
      if (!deltaY) return;
      anchoredZoom(wheelZoom(scaleRef.current, deltaY), wheelPoint.current.x, wheelPoint.current.y);
    });
  };
  el.addEventListener("wheel", onWheel, { passive: false });
  return () => {
    el.removeEventListener("wheel", onWheel);
    if (wheelRaf.current !== undefined) window.cancelAnimationFrame(wheelRaf.current);
  };
}, [anchoredZoom]);
```

```ts
const onStagePointerDown = (event: React.PointerEvent<HTMLElement>) => {
  if (event.button !== 0) return;
  if ((event.target as HTMLElement).closest(".ed-pager, .ed-warn, button, input")) return;
  const stage = stageRef.current;
  if (!stage) return;
  if (stage.scrollWidth <= stage.clientWidth + 1 && stage.scrollHeight <= stage.clientHeight + 1) return;
  panRef.current = {
    x: event.clientX,
    y: event.clientY,
    left: stage.scrollLeft,
    top: stage.scrollTop,
    moved: false,
  };
  panMoved.current = false;
  stage.setPointerCapture(event.pointerId);
  stage.classList.add("is-panning");
};

const onStagePointerMove = (event: React.PointerEvent<HTMLElement>) => {
  const pan = panRef.current;
  const stage = stageRef.current;
  if (!pan || !stage) return;
  const dx = event.clientX - pan.x;
  const dy = event.clientY - pan.y;
  if (!pan.moved && Math.hypot(dx, dy) > 4) {
    pan.moved = true;
    panMoved.current = true;
  }
  if (!pan.moved) return;
  stage.scrollLeft = pan.left - dx;
  stage.scrollTop = pan.top - dy;
};

const endPan = (event: React.PointerEvent<HTMLElement>) => {
  const pan = panRef.current;
  panRef.current = null;
  const stage = stageRef.current;
  if (!stage) return;
  stage.classList.remove("is-panning");
  if (stage.hasPointerCapture(event.pointerId)) stage.releasePointerCapture(event.pointerId);
  if (pan?.moved) window.setTimeout(() => { panMoved.current = false; }, 0);
};

const onClickCapture = (event: React.MouseEvent) => {
  if (!panMoved.current) return;
  panMoved.current = false;
  event.preventDefault();
  event.stopPropagation();
};

const onStageDoubleClick = (event: React.MouseEvent) => {
  if ((event.target as HTMLElement).closest(".tpl-page")) return;
  setFit();
};
```
